# Database Schema - Vitalbet\n\n## Users Table\n```sql\nCREATE TABLE users (\n  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),\n  email VARCHAR(255) UNIQUE NOT NULL,\n  password_hash VARCHAR(255) NOT NULL,\n  first_name VARCHAR(100),\n  last_name VARCHAR(100),\n  phone VARCHAR(20),\n  date_of_birth DATE,\n  country VARCHAR(100),\n  status VARCHAR(20) DEFAULT 'active',\n  email_verified BOOLEAN DEFAULT FALSE,\n  two_fa_enabled BOOLEAN DEFAULT FALSE,\n  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,\n  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,\n  deleted_at TIMESTAMP\n);\n\nCREATE INDEX idx_users_email ON users(email);\nCREATE INDEX idx_users_status ON users(status);\n```\n\n## User Accounts Table\n```sql\nCREATE TABLE user_accounts (\n  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),\n  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,\n  balance DECIMAL(15, 2) DEFAULT 0.00,\n  available_balance DECIMAL(15, 2) DEFAULT 0.00,\n  currency VARCHAR(3) DEFAULT 'USD',\n  total_deposited DECIMAL(15, 2) DEFAULT 0.00,\n  total_withdrawn DECIMAL(15, 2) DEFAULT 0.00,\n  total_winnings DECIMAL(15, 2) DEFAULT 0.00,\n  total_losses DECIMAL(15, 2) DEFAULT 0.00,\n  account_status VARCHAR(20) DEFAULT 'active',\n  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,\n  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,\n  UNIQUE(user_id)\n);\n\nCREATE INDEX idx_user_accounts_user_id ON user_accounts(user_id);\n```\n\n## Sports Events Table\n```sql\nCREATE TABLE sports_events (\n  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),\n  event_name VARCHAR(255) NOT NULL,\n  sport_type VARCHAR(50) NOT NULL,\n  league VARCHAR(100),\n  team_home VARCHAR(100),\n  team_away VARCHAR(100),\n  start_time TIMESTAMP NOT NULL,\n  end_time TIMESTAMP,\n  status VARCHAR(20) DEFAULT 'scheduled',\n  winner VARCHAR(100),\n  score_home INT,\n  score_away INT,\n  is_live BOOLEAN DEFAULT FALSE,\n  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,\n  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP\n);\n\nCREATE INDEX idx_sports_events_sport ON sports_events(sport_type);\nCREATE INDEX idx_sports_events_status ON sports_events(status);\nCREATE INDEX idx_sports_events_start_time ON sports_events(start_time);\n```\n\n## Odds Table\n```sql\nCREATE TABLE odds (\n  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),\n  event_id UUID NOT NULL REFERENCES sports_events(id) ON DELETE CASCADE,\n  bet_type VARCHAR(50) NOT NULL,\n  outcome_home DECIMAL(5, 2),\n  outcome_away DECIMAL(5, 2),\n  outcome_draw DECIMAL(5, 2),\n  over_2_5 DECIMAL(5, 2),\n  under_2_5 DECIMAL(5, 2),\n  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,\n  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP\n);\n\nCREATE INDEX idx_odds_event_id ON odds(event_id);\n```\n\n## Bets Table\n```sql\nCREATE TABLE bets (\n  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),\n  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,\n  event_id UUID NOT NULL REFERENCES sports_events(id),\n  odds_id UUID REFERENCES odds(id),\n  bet_type VARCHAR(50) NOT NULL,\n  prediction VARCHAR(100) NOT NULL,\n  stake DECIMAL(15, 2) NOT NULL,\n  odds DECIMAL(5, 2) NOT NULL,\n  potential_return DECIMAL(15, 2),\n  actual_return DECIMAL(15, 2),\n  status VARCHAR(20) DEFAULT 'pending',\n  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,\n  settled_at TIMESTAMP,\n  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP\n);\n\nCREATE INDEX idx_bets_user_id ON bets(user_id);\nCREATE INDEX idx_bets_event_id ON bets(event_id);\nCREATE INDEX idx_bets_status ON bets(status);\n```\n\n## Parlay Bets Table\n```sql\nCREATE TABLE parlay_bets (\n  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),\n  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,\n  status VARCHAR(20) DEFAULT 'pending',\n  stake DECIMAL(15, 2) NOT NULL,\n  combined_odds DECIMAL(10, 2) NOT NULL,\n  potential_return DECIMAL(15, 2),\n  actual_return DECIMAL(15, 2),\n  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,\n  settled_at TIMESTAMP,\n  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP\n);\n\nCREATE INDEX idx_parlay_bets_user_id ON parlay_bets(user_id);\nCREATE INDEX idx_parlay_bets_status ON parlay_bets(status);\n```\n\n## Parlay Bet Items Table\n```sql\nCREATE TABLE parlay_bet_items (\n  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),\n  parlay_bet_id UUID NOT NULL REFERENCES parlay_bets(id) ON DELETE CASCADE,\n  bet_id UUID NOT NULL REFERENCES bets(id),\n  item_order INT NOT NULL,\n  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP\n);\n\nCREATE INDEX idx_parlay_items_parlay_id ON parlay_bet_items(parlay_bet_id);\n```\n\n## Transactions Table\n```sql\nCREATE TABLE transactions (\n  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),\n  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,\n  transaction_type VARCHAR(20) NOT NULL,\n  amount DECIMAL(15, 2) NOT NULL,\n  currency VARCHAR(3) DEFAULT 'USD',\n  status VARCHAR(20) DEFAULT 'pending',\n  payment_method VARCHAR(50),\n  stripe_transaction_id VARCHAR(255),\n  description VARCHAR(255),\n  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,\n  completed_at TIMESTAMP,\n  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP\n);\n\nCREATE INDEX idx_transactions_user_id ON transactions(user_id);\nCREATE INDEX idx_transactions_status ON transactions(status);\nCREATE INDEX idx_transactions_type ON transactions(transaction_type);\n```\n\n## Admin Users Table\n```sql\nCREATE TABLE admin_users (\n  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),\n  user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,\n  role VARCHAR(50) NOT NULL,\n  permissions JSON,\n  is_active BOOLEAN DEFAULT TRUE,\n  last_login TIMESTAMP,\n  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,\n  updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,\n  UNIQUE(user_id)\n);\n\nCREATE INDEX idx_admin_users_role ON admin_users(role);\n```\n\n## Audit Logs Table\n```sql\nCREATE TABLE audit_logs (\n  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),\n  admin_id UUID REFERENCES admin_users(id),\n  action VARCHAR(100) NOT NULL,\n  entity_type VARCHAR(50),\n  entity_id VARCHAR(100),\n  changes JSON,\n  ip_address VARCHAR(45),\n  created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP\n);\n\nCREATE INDEX idx_audit_logs_admin_id ON audit_logs(admin_id);\nCREATE INDEX idx_audit_logs_created_at ON audit_logs(created_at);\n```\n
